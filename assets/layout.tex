\renewcommand{\tiny}{\@setfontsize\tiny{6}{8}}

    subsubsection/format={\normalfont\MLTSubsubsectionSize\sffamily\heiti\bfseries}
  }
  \@ifundefined{chapter}{}{\ctexset{chapter/format={\normalfont\MLTChapterSize\sffamily\heiti\bfseries}}}%
}
